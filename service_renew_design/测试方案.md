```python
# test_thread_safety.py
import socket
import struct
import json
import cv2
import numpy as np
import threading
import time

def send_request(host, port, req_type, image):
    """发送单个请求"""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.connect((host, port))
        
        # 编码图像
        _, img_encoded = cv2.imencode('.jpg', image)
        img_bytes = img_encoded.tobytes()
        
        # 发送头部
        header = json.dumps({
            "type": req_type,
            "image_size": len(img_bytes)
        }).encode('utf-8')
        sock.sendall(struct.pack("!I", len(header)))
        sock.sendall(header)
        sock.sendall(img_bytes)
        
        # 接收响应
        header_len = sock.recv(4)
        if len(header_len) < 4:
            return None
        header_size = struct.unpack("!I", header_len)[0]
        header_data = sock.recv(header_size)
        header = json.loads(header_data.decode('utf-8'))
        data_size = header['data_size']
        data = sock.recv(data_size)
        result = json.loads(data.decode('utf-8'))
        
        sock.close()
        return result
    except Exception as e:
        print(f"请求失败: {e}")
        return None

def test_concurrent_requests():
    """并发测试"""
    host = "127.0.0.1"
    port = 12345
    
    # 创建测试图像
    test_image = np.zeros((640, 640, 3), dtype=np.uint8)
    
    results = []
    errors = []
    
    def worker(thread_id):
        for i in range(10):
            result = send_request(host, port, "game_windows", test_image)
            if result:
                results.append((thread_id, i, result))
            else:
                errors.append((thread_id, i))
    
    # 启动10个线程
    threads = []
    for i in range(10):
        t = threading.Thread(target=worker, args=(i,))
        threads.append(t)
        t.start()
    
    # 等待所有线程完成
    for t in threads:
        t.join()
    
    print(f"成功请求: {len(results)}")
    print(f"失败请求: {len(errors)}")
    
    # 验证
    assert len(errors) == 0, f"有 {len(errors)} 个请求失败"
    assert len(results) == 100, f"应该有100个成功请求，实际{len(results)}"
    print("✅ 线程安全测试通过")

if __name__ == "__main__":
    test_concurrent_requests()
```

**通过标准：**
- ✅ 单客户端请求正常
- ✅ 10个客户端并发无错误
- ✅ 无数据竞争或崩溃
- ✅ 所有请求都返回正确结果

```python
import socket
import struct
import json
import threading

def test_connection_limit():
    """测试连接数限制"""
    connections = []
    max_connections = 100
    
    def create_connection():
        try:
            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
            sock.connect(("127.0.0.1", 12345))
            connections.append(sock)
            return True
        except:
            return False
    
    # 尝试创建超过限制的连接
    threads = []
    for i in range(max_connections + 10):
        t = threading.Thread(target=create_connection)
        threads.append(t)
        t.start()
    
    for t in threads:
        t.join()
    
    # 检查实际连接数
    active_connections = len([c for c in connections if c.fileno() != -1])
    print(f"活跃连接数: {active_connections}")
    
    # 清理
    for conn in connections:
        try:
            conn.close()
        except:
            pass
    
    assert active_connections <= max_connections, "连接数超过限制"
    print("✅ 连接数限制测试通过")
```
